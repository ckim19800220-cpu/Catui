모듈 받기 — catui.use

catui.use(['layer', 'table'], function (layer, table) { … });   // 인자 순서 = 배열 순서
catui.use('form').then(function ([form]) { … });                 // Promise도 돌려줌
catui.layer.msg('바로 써도 됩니다');                               // 전역에서는 catui.모듈로 직접 접근 가능

렌더형 모듈 — render · reload · destroy

table·laypage·tabs·dropdown·datePicker·upload·slider·rate·tree·collapse·canvas는 같은 인터페이스를 가집니다.

var inst = table.render({ elem: '#orders', id: 'orders', … });   // 인스턴스
table.reload('orders', { data: newRows });                        // 옵션 합쳐 다시 그리기
table.getThis('orders');                                          // 인스턴스 찾기
table.destroy('orders');                                          // 리스너·요청까지 정리

같은 대상에 다시 render하면 이전 인스턴스는 자동으로 정리됩니다.

이벤트 — 모듈.on('이벤트(필터)')

필터는 요소의 data-catui-filter 값(또는 id)입니다. 필터를 빼면 그 이벤트 전체를 받습니다.
