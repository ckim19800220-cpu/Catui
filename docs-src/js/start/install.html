npm · CDN

npm install @imcat-ckim/catui

import catui, { layer } from '@imcat-ckim/catui';      // ESM + TypeScript 타입
import '@imcat-ckim/catui/css';

<!-- CDN (아이콘 icons.svg는 같은 폴더에서 자동으로 찾음) -->
<link rel="stylesheet" href="https://cdn.jsdelivr.net/npm/@imcat-ckim/catui@2/dist/catui.min.css">
<script src="https://cdn.jsdelivr.net/npm/@imcat-ckim/catui@2/dist/catui.min.js"></script>

이름 정리: 제품 이름 Catui, npm 패키지 @imcat-ckim/catui, 전역 객체 window.catui, 파일 catui.*, CSS 클래스 Catui_.

전역 스크립트

<link rel="stylesheet" href="dist/catui.min.css">
<script src="dist/catui.js"></script>
<script>
catui.use(['layer', 'form'], function (layer, form) {
  layer.msg('준비되었습니다');
});
</script>

catui.use의 콜백은 문서가 준비된 뒤 실행되므로 스크립트를 <head>에 두어도 됩니다. 아이콘 스프라이트는 스크립트와 같은 폴더의 icons.svg를 자동으로 찾습니다.

ES 모듈

<script type="module">
  import catui, { layer, table, html } from './dist/catui.esm.js';
  layer.msg('ESM으로 불러왔습니다');
</script>

ESM은 전역 window.catui를 만들지 않습니다. 소스(src/js/index.js)는 빌드 없이도 바로 import할 수 있습니다.

전역 설정

catui.config({
  icons: '/static/icons.svg?v=2',   // 아이콘 스프라이트 경로를 바꿀 때
  autoInit: false,                  // data-catui-* 자동 연결을 끌 때 (스크립트 직후 호출)
});

ES 모듈과 아이콘은 file://에서 막힙니다. 정적 서버(npm run dev)로 여세요.
