컴포넌트 빌더

catui.component()로 만든 모듈은 내장 모듈과 같은 인터페이스(render · reload · getThis · destroy · on · off · set)를 자동으로 갖습니다.

var counter = catui.component({
  name: 'counter',                       // 모듈 이름 (이벤트 네임스페이스)
  config: { value: 0, step: 1 },         // 기본 옵션
  render: function (inst) {              // inst.elem · inst.config · inst.filter · inst.id
    var btn = document.createElement('button');
    btn.className = 'Catui_btn Catui_btn-md Catui_btn-secondary';
    btn.type = 'button';
    btn.textContent = String(inst.config.value);
    inst.elem.appendChild(btn);
    var onClick = function () {
      inst.config.value += inst.config.step;
      btn.textContent = String(inst.config.value);
      inst.emit('change', { value: inst.config.value });   // counter.on('change(필터)')
    };
    btn.addEventListener('click', onClick);
    inst.use(function () { btn.removeEventListener('click', onClick); btn.remove(); });   // 해제 등록
  },
});
