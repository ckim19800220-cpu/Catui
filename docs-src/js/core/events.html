문법

모듈.on('이벤트(필터)', fn)   // 필터가 같은 요소에서 난 이벤트만
모듈.on('이벤트', fn)         // 필터 상관없이 전부
var off = 모듈.on(…);  off(); // 해제
모듈.off('이벤트(필터)');     // 그 필터의 등록만 해제
모듈.off('이벤트');           // 필터 없이 부르면 그 이벤트 등록 전부 해제

콜백의 this는 이벤트가 난 요소입니다. 콜백이 false를 돌려주면 기본 동작을 막습니다(예: form 제출, table 기본 정렬).

모듈 밖에서

catui.onevent('form', 'submit(login)', fn);              // 모듈 이름으로 등록
catui.offevent('form', 'submit(login)');                 // 해제 (필터 없으면 그 이벤트 전부)
catui.event('form', 'submit', 'login', { field: {} });   // 직접 발생
