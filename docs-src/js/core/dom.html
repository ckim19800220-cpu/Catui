html · raw · escape

catui.html`<p>${name}</p>`          // { __raw } — 값은 escape
catui.html`<ul>${rows.map(r => catui.html`<li>${r}</li>`)}</ul>`   // 배열 이어 붙이기
catui.raw('<b>신뢰 마크업</b>')      // escape 없이
catui.escape('<b>')                    // '&lt;b&gt;'
catui.mount(el, catui.html`…`)          // html``은 마크업으로, 문자열은 글자로
